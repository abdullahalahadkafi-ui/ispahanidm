// Result Fetch & Render Function async function getStudentResult(event) { event.preventDefault(); const studentId = document.getElementById('search-id').value; const examType = document.getElementById('exam-type').value; // Render backend endpoint request const response = await fetch(`https://ispahanidm.onrender.com/api/results/${studentId}?exam=${examType}`); const data = await response.json(); if(data.success) { let subjectsHTML = ''; data.result.subjects.forEach(sub => { subjectsHTML += ` ${sub.name} ${sub.mcq} ${sub.cq} ${sub.total} `; }); document.getElementById('result-view-box').innerHTML = `

নাম: ${data.student.name}

গ্রেড: ${data.result.grade}

${subjectsHTML}
বিষয় MCQ CQ মোট
`; } else { alert("রেজাল্ট পাওয়া যায়নি!"); } }
